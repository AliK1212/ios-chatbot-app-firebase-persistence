// Firebase Storage Rules
// Copy these rules to your Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Allow read access to all files in the pdf_documents folder
    match /pdf_documents/{fileName} {
      allow read: if true;
    }
    
    // Allow read access to all files in the pdfs folder
    match /pdfs/{fileName} {
      allow read: if true;
    }
    
    // Allow read access to all files in any folder (for development purposes)
    // In production, you should restrict this to specific folders
    match /{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
